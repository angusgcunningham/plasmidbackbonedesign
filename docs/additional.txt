plasmidqc script

INDIR="/cs/student/projects1/aibh/2024/acunning/Projects/Results/gen_ft35kgfp"
WORKROOT="/cs/student/projects1/aibh/2024/acunning/qc_batch"
ORIV_REF="/cs/student/projects1/aibh/2024/acunning/oriV_refs.fasta"
ORIV_DB_PREFIX="/cs/student/projects1/aibh/2024/acunning/oriV_db"   # you prebuilt this with makeblastdb
SCRIPT="/cs/student/projects1/aibh/2024/acunning/Projects/Analysis/plasmidqc.py"
mkdir -p "$WORKROOT"

# script

for f in "$INDIR"/*.fasta "$INDIR"/*.fa; do
  [ -e "$f" ] || continue           # skip if no matches
  b=$(basename "$f"); b="${b%.*}"
  w="$WORKROOT/$b"
  mkdir -p "$w"

  # reuse shared BLAST DB by symlinking into the workdir
  ln -sf "${ORIV_DB_PREFIX}.nhr" "$w/oriV_db.nhr"
  ln -sf "${ORIV_DB_PREFIX}.nin" "$w/oriV_db.nin"
  ln -sf "${ORIV_DB_PREFIX}.nsq" "$w/oriV_db.nsq"

  # run plasmidqc (uses existing DB, so it won’t rebuild)
  python "$SCRIPT" \
    --fasta "$f" \
    --workdir "$w" \
    --oriv_ref "$ORIV_REF" \
    --arg_tool amrfinder
done

(
  head -n 1 "$WORKROOT"/*/plasmid_qc_summary.tsv | head -n 1
  for s in "$WORKROOT"/*/plasmid_qc_summary.tsv; do tail -n +2 "$s"; done
) > "$WORKROOT/_combined_plasmid_qc_summary.tsv"

python - <<'PY'
import glob, pandas as pd, os, sys
workroot = os.environ.get("WORKROOT", "/cs/student/projects1/aibh/2024/acunning/qc_batch")
rows=[]
for p in glob.glob(f'{workroot}/*/oriv_candidates.tsv'):
    df=pd.read_csv(p, sep='\t')
    df.insert(0,'plasmid_id',os.path.basename(os.path.dirname(p)))
    rows.append(df)
if rows:
    out=pd.concat(rows, ignore_index=True)
    out.to_csv(f'{workroot}/_combined_oriv_candidates.tsv', sep='\t', index=False)
    print('Wrote', f'{workroot}/_combined_oriv_candidates.tsv', 'rows:', len(out))
else:
    print('No candidate files found', file=sys.stderr)
PY
